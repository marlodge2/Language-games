Private Sub Buildon()
' adds on after going to
' no need to change next form
If gperfect.Value Then
' have is present
  If addcont.Value Then
  ' extra continuous is present
    If Ifvoice.Value Then
      phrase = phrase + "have been being "
      If found = 0 Then
        phrase = phrase + Regular$(mainverb)
      Else
        rules = -2
        phrase = phrase + strongperfect
      End If
    Else
      phrase = phrase + "have been " + Ing$(mainverb)
    End If
  Else
  ' have but not continuous be
    If Ifvoice.Value Then
      phrase = phrase + "have been "
      If found = 0 Then
        phrase = phrase + Regular$(mainverb)
      Else
        rules = -2
        phrase = phrase + strongperfect
      End If
    Else
      phrase = phrase + "have "
      If found = 0 Then
        phrase = phrase + Regular$(mainverb)
      Else
        rules = -2
        phrase = phrase + strongperfect
      End If
    End If
  End If
Else
' no extra have
  If addcont.Value Then
    If Ifvoice.Value Then
      phrase = phrase + "be being "
      If found = 0 Then
        phrase = phrase + Regular$(mainverb)
      Else
        rules = -2
        phrase = phrase + strongperfect
      End If
    Else
      phrase = phrase + "be " + Ing$(mainverb)
    End If
  Else
    If Ifvoice.Value Then
      phrase = phrase + "be "
      If found = 0 Then
        phrase = phrase + Regular$(mainverb)
      Else
        rules = -2
        phrase = phrase + strongperfect
      End If
    Else
      phrase = phrase + mainverb
    End If
  End If
End If

End Sub

Private Sub Buildup(frst)
' adds remainder of verb phrase to finite portion

Select Case frst
  Case 2
  ' modal or to is first element
  ' so no need to change next form
    If Ifperfect.Value Then
    ' have is present
      If continuous.Value Then
      ' continuous be is present
        If Ifvoice.Value Then
          phrase = phrase + "have been being "
          If found = 0 Then
            phrase = phrase + Regular$(mainverb)
          Else
            rules = -2
            phrase = phrase + strongperfect
          End If
        Else
          phrase = phrase + "have been " + Ing$(mainverb)
        End If
      ElseIf going.Value Then
      ' have and going to
        phrase = phrase + "have been going to "
        Buildon
      Else
      ' have but not continuous be
        If Ifvoice.Value Then
        ' passive be
          phrase = phrase + "have been "
        Else
          phrase = phrase + "have "
        End If
        If found = 0 Then
          phrase = phrase + Regular$(mainverb)
        Else
          rules = -2
          phrase = phrase + strongperfect
        End If
      End If
    Else
    ' no have
      If continuous.Value Then
        If Ifvoice.Value Then
          phrase = phrase + "be being "
          If found = 0 Then
            phrase = phrase + Regular$(mainverb)
          Else
            rules = -2
            phrase = phrase + strongperfect
          End If
        Else
          phrase = phrase + "be " + Ing$(mainverb)
        End If
      ElseIf going.Value Then
        phrase = phrase + "be going to "
        Buildon
      ElseIf Ifvoice.Value Then
        If Emphatic.Value Then
          phrase = phrase + "get "
        Else
          phrase = phrase + "be "
        End If
        If found = 0 Then
          phrase = phrase + Regular$(mainverb)
        Else
          rules = -2
          phrase = phrase + strongperfect
        End If
      Else
        phrase = phrase + mainverb
      End If
    End If
  Case 3
  ' have is first element
    If continuous.Value Then
    ' be is present
      If Ifvoice.Value Then
        phrase = phrase + "been being "
        If found = 0 Then
          phrase = phrase + Regular$(mainverb)
        Else
          rules = -2
          phrase = phrase + strongperfect
        End If
      Else
        phrase = phrase + "been " + Ing$(mainverb)
      End If
    ElseIf going.Value Then
      phrase = phrase + "been going to "
      Buildon
    Else
      If Ifvoice.Value Then phrase = phrase + "been "
      If found = 0 Then
        phrase = phrase + Regular$(mainverb)
      Else
        rules = -2
        phrase = phrase + strongperfect
      End If
    End If
  Case 4
  ' continuous or going to be is first element
    If going.Value Then
      phrase = phrase + "going to "
      Buildon
    ElseIf Ifvoice.Value Then
      phrase = phrase + "being "
      If found = 0 Then
        phrase = phrase + Regular$(mainverb)
      Else
        rules = -2
        phrase = phrase + strongperfect
      End If
    Else
      phrase = phrase + Ing$(mainverb)
    End If
  Case 5
    If found = 0 Then
      phrase = phrase + Regular$(mainverb)
    Else
      rules = -2
      phrase = phrase + strongperfect
    End If
  Case Else
    If Neg.Value Or ques.Value Then
      If mainverb <> "be" Then
        phrase = phrase + mainverb
      End If
    End If
End Select

End Sub

Private Sub Buildverb()
' assembles the verb phrase for printing
               
rules = 0
' no rule to apply yet
putin = 0
' where to put pronoun
tagputin = 0

' start by fixing tense
If Ifmodal.Value Or Emphatic.Value Then
  firstwd = 2
ElseIf Ifperfect.Value Then
  firstwd = 3
ElseIf continuous.Value Or going.Value Then
  firstwd = 4
ElseIf Ifvoice.Value Then
  firstwd = 5
Else
  firstwd = 1
End If

tagques$ = ""               ' for possible question tag
       
Select Case firstwd
  Case 2
  ' modal is filled
    If Emphatic.Value Then
      If Past.Value Then
        phrase = "did"
      ElseIf person > 4 Then
        phrase = "does"
      Else
        phrase = "do"
      End If
    Else
      phrase = modalverb(whichmodal, Abs(Past.Value))
      ' writes present or past tense form of modal
      ' to first element
    End If
    If Neg.Value Then
      tagques$ = phrase + " "
      If Past.Value Or Emphatic.Value Then
        phrase = phrase + "n't "
        If whichmodal = 5 And Not Emphatic.Value Then
          phrase = "didn't have to "
          putin = 7
          tagques$ = "did "
        End If
      Else
        Select Case whichmodal
          Case 1
            phrase = "shan't "
          Case 2
            phrase = "won't "
          Case 3
            phrase = "can't "
          Case 4
            phrase = "mayn't "
          Case 5
            phrase = "mustn't "
        End Select
      End If
    Else
      If Past.Value Then
        tagques$ = phrase + "n't "
        If whichmodal = 5 Or Emphatic.Value Then
          tagques$ = "didn't "
          putin = 4
        End If
      ElseIf Emphatic.Value Then
          tagques$ = phrase + "n't "
      Else
        Select Case whichmodal
          Case 1
            tagques$ = "shan't "
          Case 2
            tagques$ = "won't "
          Case 3
            tagques$ = "can't "
          Case 4
            tagques$ = "may not"
            tagputin = 4
          Case Else
            tagques$ = "mustn't "
        End Select
      End If
      phrase = phrase + " "
    End If
  Case 3
  ' have is filled but not modal
    If Past.Value Then
      phrase = "had"
    ElseIf person > 4 Then
      phrase = "has"
    Else
      phrase = "have"
    End If
    If Neg.Value Then
      tagques$ = phrase + " "
      phrase = phrase + "n't "
    Else
      tagques$ = phrase + "n't "
      phrase = phrase + " "
    End If
  Case Is > 3
  ' continuous or passive be is filled
  ' but not have or modal
    If Past.Value Then
      If person = 1 Or person > 4 Then
        phrase = "was"
        tagques$ = "was"
      Else
        phrase = "were"
        tagques$ = "were"
      End If
    Else
      Select Case person
        Case 1
          If Neg.Value And ques.Value Then
          ' special case for "aren't I"
            phrase = "are"
          Else
            phrase = "am"
            If Neg.Value Then
              tagques$ = "am"
            Else
              tagques$ = "are"
            End If
          End If
        Case Is > 4
          phrase = "is"
          tagques$ = "is"
        Case Else
          phrase = "are"
          tagques$ = "are"
      End Select
    End If
    If Neg.Value Then
      tagques$ = tagques$ + " "
      If person = 1 And (Stating.Value Or qtag.Value) And Not Past.Value Then
      ' special case for "I am not"
        phrase = phrase + " not "
      Else
        phrase = phrase + "n't "
      End If
    Else
      tagques$ = phrase + "n't "
      phrase = phrase + " "
      If person = 1 And qtag.Value And Not Past.Value Then
      ' special case for Aren't I
        tagques$ = "aren't "
      End If
    End If

  Case Else
  ' main verb only
    If Past.Value Then
      tagques$ = "did"
      If mainverb = "be" Then
        If person = 1 Or person > 4 Then
          phrase = "was"
          tagques$ = "was"
        Else
          phrase = "were"
          tagques$ = "were"
        End If
      ElseIf mainverb = "have" Then
        phrase = "had"
        tagques$ = "had"
      ElseIf Neg.Value Or ques.Value Then
        phrase = "did"
        tagques$ = "did"
      Else
        If found = 0 Then
          phrase = Regular$(mainverb)
          ' add a regular -ed ending
        Else
          rules = -1
          phrase = strongpast
          ' select second form of strong verb
        End If
      End If
    ElseIf mainverb = "be" Then
    ' present tense of be
      Select Case person
        Case 1
          phrase = "am"
        Case 2, 3, 4
          phrase = "are"
        Case Else
          phrase = "is"
      End Select
      tagques$ = phrase
    ElseIf person > 4 Then
    ' 3rd person singular present of other verbs
      tagques$ = "does"
      If Neg.Value Or ques.Value Then
      ' question or negative
        phrase = "does"
      Else
      ' add the 3rd-person s
        phrase = Sending$(mainverb)
      End If
    ElseIf Neg.Value Or ques.Value Then
    ' question or negative?
      phrase = "do"
      tagques$ = "do"
    Else
    ' verb is main verb
      phrase = mainverb
      tagques$ = "do"
    End If

    If Neg.Value Then
      tagques$ = tagques$ + " "
      If phrase = "am" Then
        phrase = phrase + " not "
      Else
        phrase = phrase + "n't "
      End If
    Else
      tagques$ = tagques$ + "n't "
      phrase = phrase + " "
    End If
  End Select

  Select Case person
  ' invert pronoun/verb if a question
    Case 1
      pron$ = "I"
    Case 2
      pron$ = "we"
    Case 3
      pron$ = "you"
    Case 4
      pron$ = "they"
    Case 5
      pron$ = "he"
    Case 6
      pron$ = "she"
    Case 7
      pron$ = "it"
  End Select
  
  If firstwd = 2 And whichmodal = 4 And Not Past.Value Then
  ' special case for not contracting NOT
    tagques$ = "may " + pron$ + " not "
  Else
    tagques$ = tagques$ + pron$
  End If

  If ques.Value Then
    If putin > 0 Then
      phrase = Left$(phrase, putin) + pron$ + Mid$(phrase, putin)
    Else
      phrase = phrase + pron$ + " "
    End If
  Else
    phrase = pron$ + " " + phrase
'    If tagputin > 0 Then
 '     tagques$ = Left$(tagques$, tagputin) + pron$ + Mid$(tagques$, tagputin)
  '  Else
   '   tagques$ = tagques$ + pron$
    'End If
  End If

  ' at this stage we have the first two words of the
  ' phrase and have to complete it. We can still use
  ' the firstwd variable

  Call Buildup(firstwd)

  If ques.Value Then
    phrase = phrase + " ...?"
  ElseIf qtag.Value Then
    phrase = phrase + " ..., " + tagques$ + "?"
  Else
    phrase = phrase + " ..."
  End If
  phrase = UCase$(Left$(phrase, 1)) + Mid$(phrase, 2)

  Label2.Caption = phrase

  If rules > 0 Then
    a$ = "The rule for "
    Select Case rules
      Case Is < 5
        a$ = a$ + "adding -S to " + UCase$(mainverb)
      Case 5 To 17
        a$ = a$ + "adding -ING to " + UCase$(mainverb)
      Case Else
        If Ifperfect.Value Or Ifvoice.Value Then
          a$ = a$ + "making the past participle of "
          a$ = a$ + UCase$(mainverb)
        Else
          a$ = a$ + "making " + UCase$(mainverb) + " past"
        End If
    End Select
    a$ = a$ + " is: " + rule$(rules) + "."
    Label3.Caption = a$
    Label3.Visible = True
  ElseIf rules < 0 Then
    a$ = UCase$(mainverb) + " is an irregular verb whose "
    If rules = -1 Then
      a$ = a$ + "past tense form is "
      a$ = a$ + UCase$(strongpast) + "."
    Else
      a$ = a$ + "past participle form is "
      a$ = a$ + UCase$(strongperfect) + "."
    End If
    Label3.Caption = a$
    Label3.Visible = True
  Else
    Label3.Visible = False
  End If

  If intransitive And Ifvoice.Value Then
    z$ = "This intransitive verb is not normally used in the passive."
  Else
    z$ = ""
  End If
  If stative And (continuous.Value Or addcont.Value) Then
    Y$ = "This stative verb is not normally used in continuous forms.  "
  ElseIf Not intransitive Then
    Y$ = ""
  End If

  If whichmodal = 5 And Ifmodal.Value And Neg.Value Then
    If Present.Value Then
      zz$ = "The negative of MUST is a prohibition. Is this what you mean? Otherwise, use HAVE TO."
    Else
      zz$ = "The negative of HAVE TO means no obligation. Is this what you mean? Otherwise use COULDN'T."
    End If
  Else
    zz$ = ""
  End If

  Shownote.Caption = note
  warnings.Caption = Y$ + z$ + zz$

End Sub

Private Sub Discover()
' search for verb in strong verb array

  If strongtotal = 0 Then Exit Sub
  increment = strongtotal \ 4
  pointer = strongtotal \ 2
  found = 0
  retried = 0

  Do
    If pointer > strongtotal Then
      pointer = strongtotal
    ElseIf pointer < 1 Then
      pointer = 1
    End If
    Get #1, pointer, target
    If mainverb = LTrim$(RTrim$(target.spell)) Then
      found = pointer
      Exit Do
    ElseIf mainverb > LTrim$(RTrim$(target.spell)) Then
      pointer = pointer + increment
    Else
      pointer = pointer - increment
    End If
    increment = increment \ 2 + increment Mod 2
    If increment < 2 Then
      retried = retried + 1
    End If
    If retried > 3 Then Exit Do
  Loop
  If found Then
    If (target.limits And 1) = 1 And (target.limits And 2) <> 2 Then
      ' user has entered oblique tense of a strong verb
      Msg$ = "Do you really mean TO " + UCase$(mainverb) + "?"
      meantit = MsgBox(Msg$, 292, "SURE?")
      If meantit = 6 Then
        Label4.Caption = "Unrecognised verb"
        intransitive = False
        stative = False
        found = 0
      Else
        mainverb = oldmainverb
        Exit Sub
      End If
    Else
      Label4.Caption = target.defin
      If target.limits And 8 Then
        intransitive = True
      Else
        intransitive = False
      End If
      If (target.limits And 64) Or (target.limits And 128) Then
        stative = True
      Else
        stative = False
      End If
    End If
  ElseIf Right$(UCase$(mainverb), 2) = "ED" Then
    Msg$ = "Do you really mean TO " + UCase$(mainverb) + "?"
    meantit = MsgBox(Msg$, 292, "SURE?")
    If meantit = 6 Then
      Label4.Caption = "Unrecognised verb"
      intransitive = False
      stative = False
    Else
      mainverb = oldmainverb
      Exit Sub
    End If
  Else
    For z = 1 To 5
      For zz = 0 To 1
        If mainverb = modalverb(z, zz) Then
        ' user has entered a modal
          Msg$ = "Do you really mean TO " + UCase$(mainverb) + "?"
          meantit = MsgBox(Msg$, 292, "SURE?")
          If meantit = 6 Then
            Exit For
          Else
            mainverb = oldmainverb
            Exit Sub
          End If
        End If
      Next zz
    Next z
    Label4.Caption = "Unrecognised verb"
    intransitive = False
    stative = False
  End If
  note = ""
  If found And target.reclong > 1 Then
  ' a note exists
    note = Space$(target.reclong)
    Get #2, target.recpoint, note
    If target.limits And 1 Then
    ' this is a strong verb
      chop = InStr(note, " ")
      ' look for first word of note
      If chop > 0 Then strongpast = Left$(note, chop - 1)
      ' and put it into past tense
      note = Mid$(note, chop + 1)
      ' rest of note
      chop = InStr(note, "\")
      ' look for next element up to \
      If chop > 0 Then strongperfect = Left$(note, chop - 1)
      ' and assign it to past participle
      note = Mid$(note, chop + 1)
      ' rest of note after \
    Else
      found = 0
      ' not a strong verb
    End If
  Else
    found = 0
    ' no note and not a strong verb
  End If

  current.Caption = " " + mainverb + " "
  oldmainverb = mainverb
  Buildverb

End Sub

Private Function Hasaprefix(a$)
' returns TRUE if a$ begins with a Latin prefix
  Hasaprefix = False
  If Len(a$) > 3 And syllable < 3 Then
    If Left$(a$, 2) = "re" Then
      c = Isavowel(Mid$(a$, 3, 1))
      cc = Isavowel(Mid$(a$, 4, 1))
      If (c Or cc) And a$ <> "reason" Then Hasaprefix = True
      Exit Function
    End If
    check = InStr("ab/ad/be/de/ex/in/im/oc/un", Left$(a$, 2))
    If check > 0 And a$ <> "exit" Then
      Hasaprefix = True
      Exit Function
    End If
    If Len(a$) > 4 And syllable < 3 Then
      check = InStr("con/com/for/mis/out/pre/per/sub", Left$(a$, 3))
      If check > 0 Then
        Hasaprefix = True
        Exit Function
      End If
    End If
    If Len(a$) > 5 Then
      check = InStr("over", Left$(a$, 4))
      If check > 0 Then
        Hasaprefix = True
        Exit Function
      End If
    End If
    If Len(a$) > 6 Then
      check = InStr("handi/inter/super/trans/under", Left$(a$, 5))
      If check > 0 Then Hasaprefix = True
    End If
  End If
End Function

Private Function Ing$(a$)
' adds -ing to a

X$ = Right$(a$, 1)
' last letter of word
If Len(a$) > 1 Then
  Y$ = Mid$(a$, Len(a$) - 1, 1)
  ' second-last letter
Else
' one letter only entered
  Ing$ = UCase$(a$) + "ing"
  Exit Function
End If
If Len(a$) > 2 Then
  z$ = Mid$(a$, Len(a$) - 2, 1)
Else
  z$ = "a"
End If
' third-last letter
If Len(a$) > 3 Then
  zz$ = Mid$(a$, Len(a$) - 3, 1)
Else
  zz$ = "a"
End If
' fourth-last letter

Select Case X$
  Case "e"
  ' word ends in -e
  If Len(a$) < 3 Then
  ' verb to be
    Ing$ = a$ + "ing"
    rules = 5
  ElseIf Y$ = "i" Then
  ' word ends in ie
    Ing$ = Left$(a$, Len(a$) - 2) + "ying"
    rules = 6
  ElseIf Y$ = "e" Or Y$ = "o" Or Y$ = "y" Then
  ' word ends in double e, oe or ye
    Ing$ = a$ + "ing"
    rules = 7
  ElseIf Y$ = "g" And z$ = "n" And zz$ = "i" Then
  ' verb ends in inge
    Ing$ = a$ + "ing"
    rules = 7
  ElseIf Y$ = "r" And Not Isavowel(z$) Then
  ' word ends in consonant + re
    Ing$ = Left$(a$, Len(a$) - 2) + "ering"
    rules = 34
  Else
  ' all other letter + e endings
    Ing$ = Left$(a$, Len(a$) - 1) + "ing"
    rules = 8
    ' so drop the final e
  End If
           
  Case "c"
  ' ends in c
  If Isavowel(Y$) And Not Isavowel(z$) Then
  ' eg picnic/picnicking
    Ing$ = a$ + "king"
    rules = 9
  Else
    Ing$ = a$ + "ing"
    rules = 10
  End If
           
  Case "a", "f", "h", "i", "j", "o", "q", "u", "w", "x", "y"
  ' single-f, j and q only possible in loan words
  Ing$ = a$ + "ing"
  rules = 11
           
  Case Else
  ' b, d, g, k, l, m, n, p, r, s, t, v, and z
  If Isavowel(Y$) And (Not Isavowel(z$) Or zz$ = "q") Then
  ' word ends in one consonant preceded by one vowel
    Select Case X$
      Case "r"
      ' word ends in r
      If syllable > 2 Then
        Ing$ = a$ + "ing"
        rules = 12
      ElseIf Hasaprefix(a$) Or a$ <> "conquer" Then
      ' stress on second syllable so double the last letter
        Ing$ = a$ + Right$(a$, 1) + "ing"
        rules = 13
      ElseIf syllable = 1 Then
      ' words like stir, bar, etc which double the -r
        Ing$ = a$ + Right$(a$, 1) + "ing"
        rules = 14
      Else
      ' words like offer and cater
        Ing$ = a$ + "ing"
        rules = 16
      End If
                  
      Case "n", "t"
      If syllable > 2 And Not Hasaprefix(a$) Or a$ = "interpret" Then
        Ing$ = a$ + "ing"
        rules = 12
      ElseIf syllable = 2 Then
      ' word ends in vowel+ n or vowel+t
        If Hasaprefix(a$) Or a$ = "emit" Or a$ = "omit" Then
        ' stress on second syllable so double the last letter
          Ing$ = a$ + Right$(a$, 1) + "ing"
          rules = 13
        Else
        ' words like visit, audit, listen
          Ing$ = a$ + "ing"
          rules = 16
        End If
      Else
      ' word is one-syllable so double final consonant
        Ing$ = a$ + Right$(a$, 1) + "ing"
        rules = 14
      End If
      Case "l"
      If syllable > 2 Then
        Ing$ = a$ + "ing"
        rules = 12
      ElseIf syllable = 1 Then
        Ing$ = a$ + Right$(a$, 1) + "ing"
        rules = 14
      Else
        Ing$ = a$ + Right$(a$, 1) + "ing"
        rules = 15
      End If
      Case "p"
      If syllable > 2 And Not Hasaprefix(a$) Then
        Ing$ = a$ + "ing"
        rules = 12
      ElseIf syllable = 1 Then
        Ing$ = a$ + Right$(a$, 1) + "ing"
        rules = 14
      ElseIf Hasaprefix(a$) Or a$ = "equip" Then
        Ing$ = a$ + Right$(a$, 1) + "ing"
        ' double the last letter
        rules = 13
      Else
        Ing$ = a$ + "ing"
        rules = 16
      End If

      Case Else
      ' other than r, l, t or n
      If syllable > 2 And Not Hasaprefix(a$) Then
        Ing$ = a$ + "ing"
        rules = 12
      Else
        Ing$ = a$ + Right$(a$, 1) + "ing"
        rules = 17
        ' so double the last letter
      End If
      End Select
    Else
      Ing$ = a$ + "ing"
      rules = 10
    End If
           
  End Select

End Function

Private Function Regular$(a$)
' adds ed ending to weak verbs for past tense or participle
         
X$ = Right$(a$, 1)
If Len(a$) > 1 Then
  Y$ = Mid$(a$, Len(a$) - 1, 1)
  ' second last letter
Else
' single letter verb
  Regular$ = UCase$(a$) + "ed"
  Exit Function
End If
If Len(a$) > 2 Then
  z$ = Mid$(a$, Len(a$) - 2, 1)
Else
  z$ = "a"
End If
' third last letter
          
If a$ = "be" Then
  If Ifperfect.Value Or Ifvoice.Value Then
    Regular$ = "been"
    rules = 18
  ElseIf person = 1 Or person = 4 Then
    rules = 19
    Regular$ = "was"
  Else
    Regular$ = "were"
    rules = 20
  End If
  Exit Function
End If
      
Select Case X$
  Case "e"
    If Y$ = "r" And Not Isavowel(z$) Then
    ' word ends in consonant+re
      Regular$ = Left$(a$, Len(a$) - 2) + "ered"
      rules = 35
    Else
      Regular$ = a$ + "d"
      rules = 21
    End If
  Case "y"
    If Not Isavowel(Y$) Then
      Regular$ = Left$(a$, Len(a$) - 1) + "ied"
      rules = 22
    Else
      Regular$ = a$ + "ed"
      rules = 23
    End If
  Case "a", "f", "h", "i", "j", "o", "q", "u", "w", "x"
    Regular$ = a$ + "ed"
    rules = 24
  Case Else
  ' b, c, d, g, k, l, m, n, p, r, s, t, v, and z
    If Isavowel(Y$) And (Not Isavowel(z$) Or zz$ = "q") Then
    ' ends in a single consonant preceded by single vowel
      Select Case X$
        Case "r"
        ' word ends in r
          If Y$ = "e" Then
          ' word ends in -er
            If syllable > 2 Then
            ' 3-syllables, no doubling
               Regular$ = a$ + "ed"
               rules = 25
            ElseIf Hasaprefix(a$) Or a$ <> "conquer" Then
            ' stress on second syllable so double the last letter
              Regular$ = a$ + Right$(a$, 1) + "ed"
              rules = 26
            Else
            ' words like offer, utter, cater
              Regular$ = a$ + "ed"
              rules = 27
            End If
          Else
            Regular$ = a$ + Right$(a$, 1) + "ed"
            rules = 28
            ' double the r
          End If
      
      Case "n", "t"
        If syllable > 2 And Not Hasaprefix(a$) Or a$ = "interpret" Then
        ' three-syllable word
          Regular$ = a$ + "ed"
          rules = 25
        ElseIf syllable = 2 Then
          If Hasaprefix(a$) Or a$ = "emit" Or a$ = "omit" Then
          ' Latin prefix so stress last syllable, double consonant
            Regular$ = a$ + Right$(a$, 1) + "ed"
            rules = 26
          Else
          ' words like visit, listen, audit
            Regular$ = a$ + "ed"
            rules = 27
          End If
        Else
        ' word is one-syllable so double final consonant
          Regular$ = a$ + Right$(a$, 1) + "ed"
          rules = 29
        End If
      Case "c"
      ' eg frolic, panic, picnic, shellac
        Regular$ = a$ + "ked"
        rules = 30
      Case "l"
        If syllable > 2 Then
          Regular$ = a$ + "ed"
          rules = 25
        Else
          Regular$ = a$ + Right$(a$, 1) + "ed"
          rules = 31
        End If
      Case Else
      ' b, d, g, j, k, m, p, s, v, z
      ' preceded by one vowel but not two
        Regular$ = a$ + Right$(a$, 1) + "ed"
        rules = 32
        ' so double the last letter
    End Select
  Else
    Regular$ = a$ + "ed"
    rules = 33
  End If
End Select

End Function

Private Function Sending$(a$)
' adds -s to a verb

If a$ = "have" Then
  Sending$ = "has"
  rules = 36
  Exit Function
End If

X$ = Right$(a$, 1)
' last letter
If Len(a$) > 1 Then
  Y$ = Mid$(a$, Len(a$) - 1, 1)
  ' second last letter
Else
' one letter only entered
  Sending$ = UCase$(a$) + "s"
  Exit Function
End If
If Len(a$) > 2 Then
  z$ = Mid$(a$, Len(a$) - 2, 1)
Else
  z$ = ""
End If
' third-last letter
If Len(a$) > 3 Then
  zz$ = Mid$(a$, Len(a$) - 3, 1)
Else
  z$ = ""
End If
' fourth-last letter


Select Case X$
  Case "s", "z"
    If Isavowel(Y$) And (Not Isavowel(z$) Or zz$ = "q") Then
      If syllable < 3 Then
        Sending$ = a$ + Right$(a$, 1) + "es"
        ' double the s after a single vowel
        rules = 1
      Else
        Sending$ = a$ + "es"
        rules = 2
      End If
    Else
      Sending$ = a$ + "es"
      rules = 2
    End If
  Case "x"
    Sending$ = a$ + "es"
    rules = 2
  Case "h"
    If Y$ = "c" And z$ <> "i" And (Isavowel(z$) Or z$ = "y") And Not Isavowel(zz$) Then
      Sending$ = a$ + "s"
      rules = 37
    ElseIf Y$ = "c" Or Y$ = "s" Then
      Sending$ = a$ + "es"
      rules = 2
    Else
      Sending$ = a$ + "s"
      rules = 4
    End If
  Case "y"
    If Not Isavowel(Y$) Then
      Sending$ = Left$(a$, Len(a$) - 1) + "ies"
      rules = 3
    Else
      Sending$ = a$ + "s"
      rules = 4
    End If
  Case "o"
    If Not Isavowel(Y$) Then
      Sending$ = a$ + "es"
      rules = 2
    Else
      Sending$ = a$ + "s"
      rules = 4
    End If
  Case Else
    Sending$ = a$ + "s"
    rules = 4
End Select

End Function

Private Sub Writerule()
rule(1) = "double the S and add ES if the verb has less than three syllables and ends in one vowel plus S or -Z"
rule(2) = "add ES if the verb ends in S, X, Z, SH, CH or consonant + O"
rule(3) = "change Y to IES if the verb ends in consonant plus Y"
rule(4) = "add S to verbs which do not end in S, SH, CH, consonant + Y or consonant + O"
